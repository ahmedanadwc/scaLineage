scaLineage
Custom built SAS macro programs to trigger recording and extracting information about input, output, and the use of macro symbols from a SAS program while it is running. The captured information can provide a simple static data lineage diagram. As a minimum, Base SAS Software, version 9.4 Maintenance 4 installation is required to run these macros.

1/**
2 @file lin_buildLineage.sas
3 @brief Resolves the column level lineage of one SCAPROC recording.
4 @details Needs the tables generated by
5
6 - %lin_readLog
7 - %lin_prepSource
8 - %lin_exprDataStep
9 - %lin_exprSql
10 - %lin_tokenize
11
12 SCAPROC records which columns every step read and wrote (ATTR records). Those
13 facts are combined with the derivations found in the step source, so every column written by a step
14 gets one or more edges back to the column(s) it came from:
15|From|Description|
16|---|---|
17|EXPR|derived by an expression (assignment / select item); EXPR holds the text|
18|RENAME|renamed (rename= option or RENAME statement)|
19|NAME|same column name in an input of the step (pass-through; also the only information available for steps generated by a macro, which have no recorded source)|
20|FILE|read from an external file (PROC IMPORT, INFILE, ...)|
21|INPUT|read by an INPUT statement from an INFILE or in-stream data|
22|CONST|expression without any column (constant, COUNT(*), ...)|
23|AUTO|automatic column of PROC SUMMARY / MEANS (_FREQ_, _TYPE_)|
24|NOSOURCE|the step has no recorded source (code that was %included or generated by a macro that was itself %included, CALL EXECUTE, ...) and the column name is in no input, so nothing can be said|
25|UNRESOLVED|the step has source, but nothing in it says where the column came from|
26
27
28 A data set that is written more than once gets a version number (1, 2, ...), so a step that reads
29 and rewrites the same table (PROC SORT, PROC APPEND, ...) does not create a cycle. Version 0 is a
30 data set that was only read (it already existed).
31
32 Creates
33 <ul><li>work.&p_prefix._collin: Column level lineage for each step</li>
34 |variable|Description|
35 |---|---|
36 |step|The step number of the DATA step|
37 |procname|The name of the procedure|
38 |method|The method used by the procedure|
39 |fromKind|The kind of the source <br>(data set (ds), file, inline, none)|
40 |fromDs|The source data set|
41 |fromVer|The version of the source data set|
42 |fromVar|The source variable|
43 |toDs|The target data set|
44 |toVer|The version of the target data set|
45 |toVar|The target variable|
46 |expr|The expression that defines the target variable|
47
48 <li>work.&p_prefix._dsver: Dataset version information for each step</li>
49 |variable|Description|
50 |---|---|
51 |step|The step number of the DATA step|
52 |io(IN\|OUT)|The role of the data set (input or output)|
53 |ds|The data set|
54 |ver|The version of the data set|
55 </ul>

115 /* PROC IMPORT (and similar) read the file with a generated DATA step that SCAPROC also records as
116 reading the table it is creating. Such a "ghost" input is dropped: the table comes from the file. */
117 PROC SQL;
118 CREATE TABLE work._lin_ghost AS
119 SELECT DISTINCT i.step, i.ds
120 FROM &p_dsioDsName i
121 INNER JOIN &p_dsioDsName o ON o.step = i.step AND o.ds = i.ds AND o.role = 'OUTPUT'
122 WHERE i.role = 'INPUT'
123 AND i.step IN (SELECT step FROM &p_fileioDsName WHERE role = 'INPUT')
124 AND NOT EXISTS (SELECT 1 FROM &p_dsioDsName w WHERE w.ds = i.ds AND w.role IN ('OUTPUT', 'UPDATE') AND w.step < i.step);
125
126 DELETE FROM &p_dsioDsName a
127 WHERE a.role = 'INPUT' AND EXISTS (SELECT 1 FROM work._lin_ghost g WHERE g.step = a.step AND g.ds = a.ds);
128 DELETE FROM &p_attrsDsName a
129 WHERE a.role = 'INPUT' AND EXISTS (SELECT 1 FROM work._lin_ghost g WHERE g.step = a.step AND g.ds = a.ds);
130 QUIT;

132 /* ---- attributes: drop repeated records, number the output columns of every data set ---- */
133 PROC SORT DATA=&p_attrsDsName OUT=work._lin_a NODUPKEY;
134 BY step role ds var;
135 RUN;
136 PROC SORT DATA=work._lin_a;
137 BY step role ds ord;
138 RUN;
139 DATA work._lin_a;
140 SET work._lin_a;
141 BY step role ds;
142 LENGTH ukey $64;
143 RETAIN seq 0;
144 IF (FIRST.ds) THEN seq = 0;
145 seq + 1;
146 ukey = UPCASE(var);
147 RUN;

149 /* ---- data set names written by the SQL source -> the LIB.MEM names SCAPROC recorded ---- */
150 PROC SQL NOPRINT;
151 CREATE TABLE work._lin_dsmem AS
152 SELECT DISTINCT step, ds, SCAN(ds, 2, '.') AS mem LENGTH=64
153 FROM &p_dsioDsName;
154
155 CREATE TABLE work._lin_raw AS
156 SELECT DISTINCT step, tblRaw AS raw LENGTH=200 FROM &p_aliasDsName WHERE tblRaw NE '(subquery)'
157 UNION
158 SELECT DISTINCT step, toRaw FROM &p_eexprDsName WHERE NOT MISSING(toRaw);
159
160 CREATE TABLE work._lin_nm AS
161 SELECT DISTINCT r.step, r.raw
162 , COALESCE(d.ds, CASE WHEN INDEX(r.clean, '.') > 0 THEN r.clean ELSE CATS('WORK.', r.clean) END) AS ds LENGTH=80
163 FROM (SELECT step, raw, UPCASE(STRIP(SCAN(raw, 1, '('))) AS clean LENGTH=200 FROM work._lin_raw) r
164 LEFT JOIN work._lin_dsmem d
165 ON d.step = r.step AND (d.ds = r.clean OR (INDEX(r.clean, '.') = 0 AND d.mem = r.clean));
166
167 CREATE TABLE work._lin_al AS
168 SELECT a.step, a.qid, a.alias, n.ds LENGTH=80
169 FROM &p_aliasDsName a
170 LEFT JOIN work._lin_nm n ON n.step = a.step AND n.raw = a.tblRaw;

172 /* ---- which output columns does each expression define? ---- */
173 CREATE TABLE work._lin_tgt AS
174 SELECT e.eid, o.ds AS toDs LENGTH=80, o.var AS toVar LENGTH=64
175 FROM &p_eexprDsName e
176 INNER JOIN work._lin_a o ON o.step = e.step AND o.role = 'OUTPUT' AND o.ukey = UPCASE(e.toVar)
177 WHERE e.qid = 0
178 UNION
179 SELECT e.eid, o.ds, o.var
180 FROM &p_eexprDsName e
181 INNER JOIN work._lin_nm n ON n.step = e.step AND n.raw = e.toRaw
182 INNER JOIN work._lin_a o ON o.step = e.step AND o.role = 'OUTPUT' AND o.ds = n.ds
183 AND ((NOT MISSING(e.toVar) AND o.ukey = UPCASE(e.toVar)) OR (MISSING(e.toVar) AND o.seq = e.pos))
184 WHERE e.qid > 0;

186 /* ---- rename= option: the input data set it is attached to (when that is one of the step's inputs) ---- */
187 CREATE TABLE work._lin_rnin AS
188 SELECT DISTINCT r.eid, i.ds LENGTH=80
189 FROM &p_refDsName r
190 INNER JOIN &p_eexprDsName e ON e.eid = r.eid AND e.kind = 'RENAME'
191 INNER JOIN work._lin_a i ON i.step = e.step AND i.role = 'INPUT'
192 AND (i.ds = UPCASE(STRIP(r.qual))
193 OR (INDEX(r.qual, '.') = 0 AND SCAN(i.ds, 2, '.') = UPCASE(STRIP(r.qual))))
194 WHERE NOT MISSING(r.qual);
195
196 /* ---- which input columns does each reference point to? ---- */
197 CREATE TABLE work._lin_src0 AS
198 /* DATA step: any input column with that name (a rename= option only the data set it belongs to) */
199 SELECT r.eid, r.ident LENGTH=64, i.ds AS fromDs LENGTH=80, i.var AS fromVar LENGTH=64
200 FROM &p_refDsName r
201 INNER JOIN &p_eexprDsName e ON e.eid = r.eid
202 INNER JOIN work._lin_a i ON i.step = e.step AND i.role = 'INPUT' AND i.ukey = UPCASE(r.ident)
203 WHERE e.qid = 0
204 AND (NOT EXISTS (SELECT 1 FROM work._lin_rnin x WHERE x.eid = r.eid)
205 OR EXISTS (SELECT 1 FROM work._lin_rnin x WHERE x.eid = r.eid AND x.ds = i.ds))
206 UNION
207 /* SQL: alias.column with a known table */
208 SELECT r.eid, r.ident, i.ds, i.var
209 FROM &p_refDsName r
210 INNER JOIN &p_eexprDsName e ON e.eid = r.eid
211 INNER JOIN work._lin_al a ON a.step = e.step AND a.qid = e.qid AND a.alias = UPCASE(r.qual) AND NOT MISSING(a.ds)
212 INNER JOIN work._lin_a i ON i.step = e.step AND i.role = 'INPUT' AND i.ds = a.ds AND i.ukey = UPCASE(r.ident)
213 WHERE e.qid > 0 AND NOT MISSING(r.qual) AND r.calc = 0
214 UNION
215 /* SQL: alias.column where the alias is a sub-query / unknown: any input column with that name */
216 SELECT r.eid, r.ident, i.ds, i.var
217 FROM &p_refDsName r
218 INNER JOIN &p_eexprDsName e ON e.eid = r.eid
219 INNER JOIN work._lin_a i ON i.step = e.step AND i.role = 'INPUT' AND i.ukey = UPCASE(r.ident)
220 WHERE e.qid > 0 AND NOT MISSING(r.qual) AND r.calc = 0
221 AND NOT EXISTS (SELECT 1 FROM work._lin_al a
222 WHERE a.step = e.step AND a.qid = e.qid AND a.alias = UPCASE(r.qual) AND NOT MISSING(a.ds))
223 UNION
224 /* SQL: plain column, looked for in the tables of its query segment (or in every input if none are known) */
225 SELECT r.eid, r.ident, i.ds, i.var
226 FROM &p_refDsName r
227 INNER JOIN &p_eexprDsName e ON e.eid = r.eid
228 INNER JOIN work._lin_a i ON i.step = e.step AND i.role = 'INPUT' AND i.ukey = UPCASE(r.ident)
229 WHERE e.qid > 0 AND MISSING(r.qual) AND r.calc = 0
230 AND (NOT EXISTS (SELECT 1 FROM work._lin_al a WHERE a.step = e.step AND a.qid = e.qid AND NOT MISSING(a.ds))
231 OR i.ds IN (SELECT a.ds FROM work._lin_al a WHERE a.step = e.step AND a.qid = e.qid));
232
233 /* ---- references to a column that was itself derived in the same step / query segment ---- */
234 CREATE TABLE work._lin_dr AS
235 SELECT DISTINCT r.eid, f.eid AS depEid
236 FROM &p_refDsName r
237 INNER JOIN &p_eexprDsName e ON e.eid = r.eid
238 INNER JOIN &p_eexprDsName f ON f.step = e.step AND f.qid = e.qid AND f.kind = 'EXPR'
239 AND UPCASE(f.toVar) = UPCASE(r.ident) AND f.eid NE r.eid
240 WHERE e.kind = 'EXPR'
241 AND NOT EXISTS (SELECT 1 FROM work._lin_src0 s WHERE s.eid = r.eid AND s.ident = r.ident);
242
243 CREATE TABLE work._lin_res AS
244 SELECT DISTINCT eid, fromDs, fromVar FROM work._lin_src0;
245 QUIT;

253 CREATE TABLE work._lin_new AS
254 SELECT DISTINCT d.eid, s.fromDs, s.fromVar
255 FROM work._lin_dr d
256 INNER JOIN work._lin_res s ON s.eid = d.depEid

265 PROC SQL NOPRINT;
266 /* ---- edges found from the source ---- */
267 CREATE TABLE work._lin_eng AS
268 SELECT e.step, t.toDs, t.toVar, e.kind AS method LENGTH=10, 'DS' AS fromKind LENGTH=6
269 , s.fromDs LENGTH=1000, s.fromVar LENGTH=64, SUBSTR(e.expr, 1, 500) AS expr LENGTH=500
270 FROM &p_eexprDsName e
271 INNER JOIN work._lin_tgt t ON t.eid = e.eid
272 INNER JOIN work._lin_res s ON s.eid = e.eid
273 WHERE e.kind IN ('EXPR', 'RENAME')
274 UNION
275 SELECT e.step, t.toDs, t.toVar, 'CONST', 'NONE', '', '', SUBSTR(e.expr, 1, 500)
276 FROM &p_eexprDsName e
277 INNER JOIN work._lin_tgt t ON t.eid = e.eid
278 WHERE e.kind = 'EXPR' AND NOT EXISTS (SELECT 1 FROM work._lin_res s WHERE s.eid = e.eid)
279 AND NOT EXISTS (SELECT 1 FROM &p_refDsName r WHERE r.eid = e.eid)
280 UNION
281 SELECT e.step, t.toDs, t.toVar, 'INPUT'
282 , CASE WHEN f.path IS NULL THEN 'INLINE' ELSE 'FILE' END
283 , COALESCE(f.path, '(in-stream data)'), '', ''
284 FROM &p_eexprDsName e
285 INNER JOIN work._lin_tgt t ON t.eid = e.eid
286 LEFT JOIN (SELECT DISTINCT step, path FROM &p_fileioDsName WHERE role = 'INPUT') f ON f.step = e.step
287 WHERE e.kind = 'INPUT';

289 /* ---- everything else: same column name in an input of the step ---- */
290 CREATE TABLE work._lin_name AS
291 SELECT o.step, o.ds AS toDs LENGTH=80, o.var AS toVar LENGTH=64, 'NAME' AS method LENGTH=10, 'DS' AS fromKind LENGTH=6
292 , i.ds AS fromDs LENGTH=1000, i.var AS fromVar LENGTH=64, '' AS expr LENGTH=500
293 FROM work._lin_a o
294 INNER JOIN work._lin_a i ON i.step = o.step AND i.role = 'INPUT' AND i.ukey = o.ukey
295 WHERE o.role = 'OUTPUT'
296 AND NOT EXISTS (SELECT 1 FROM work._lin_eng g WHERE g.step = o.step AND g.toDs = o.ds AND g.toVar = o.var);

298 CREATE TABLE work._lin_c0 AS
299 SELECT * FROM work._lin_eng
300 UNION ALL
301 SELECT * FROM work._lin_name;

303 /* ---- output columns still without a source: read from a file, or unresolved ---- */
304 CREATE TABLE work._lin_miss AS
305 SELECT o.step, o.ds AS toDs LENGTH=80, o.var AS toVar LENGTH=64
306 FROM work._lin_a o
307 WHERE o.role = 'OUTPUT'
308 AND NOT EXISTS (SELECT 1 FROM work._lin_c0 g WHERE g.step = o.step AND g.toDs = o.ds AND g.toVar = o.var);
309
310 CREATE TABLE work._lin_file AS
311 SELECT m.step, m.toDs, m.toVar, 'FILE' AS method LENGTH=10, 'FILE' AS fromKind LENGTH=6
312 , f.path AS fromDs LENGTH=1000, '' AS fromVar LENGTH=64, '' AS expr LENGTH=500
313 FROM work._lin_miss m
314 INNER JOIN (SELECT DISTINCT step, path FROM &p_fileioDsName WHERE role = 'INPUT') f ON f.step = m.step;
315
316 CREATE TABLE work._lin_unres AS
317 SELECT m.step, m.toDs, m.toVar
318 , CASE WHEN m.toVar IN ('_FREQ_', '_TYPE_') AND p.procname IN ('SUMMARY', 'MEANS') THEN 'AUTO'
319 WHEN NOT EXISTS (SELECT 1 FROM &p_stmtsDsName t WHERE t.step = m.step) THEN 'NOSOURCE'
320 ELSE 'UNRESOLVED' END AS method LENGTH=10
321 , 'NONE' AS fromKind LENGTH=6
322 , '' AS fromDs LENGTH=1000, '' AS fromVar LENGTH=64, '' AS expr LENGTH=500
323 FROM work._lin_miss m
324 LEFT JOIN &&p_stepsDsName p ON p.step = m.step
325 WHERE NOT EXISTS (SELECT 1 FROM work._lin_file x WHERE x.step = m.step AND x.toDs = m.toDs AND x.toVar = m.toVar);
326
327 CREATE TABLE work._lin_c1 AS
328 SELECT * FROM work._lin_c0
329 UNION ALL SELECT * FROM work._lin_file
330 UNION ALL SELECT * FROM work._lin_unres;

332 /* ---- data set versions: every step that writes a data set creates its next version ---- */
333 CREATE TABLE work._lin_writes AS
334 SELECT DISTINCT step, ds FROM &p_dsioDsName WHERE role IN ('OUTPUT', 'UPDATE');
335
336 CREATE TABLE &p_prefix._dsver AS
337 SELECT DISTINCT d.step, 'IN' AS io LENGTH=3, d.ds LENGTH=80
338 , (SELECT COUNT(*) FROM work._lin_writes w WHERE w.ds = d.ds AND w.step < d.step) AS ver
339 FROM &p_dsioDsName d WHERE d.role IN ('INPUT', 'UPDATE')
340 UNION
341 SELECT DISTINCT d.step, 'OUT', d.ds
342 , (SELECT COUNT(*) FROM work._lin_writes w WHERE w.ds = d.ds AND w.step <= d.step)
343 FROM &p_dsioDsName d WHERE d.role IN ('OUTPUT', 'UPDATE');
344
345 CREATE TABLE &p_prefix._collin AS
346 SELECT c.step, s.procname LENGTH=32, c.method, c.fromKind, c.fromDs
347 , CASE WHEN c.fromKind = 'DS'
348 THEN (SELECT COUNT(*) FROM work._lin_writes w WHERE w.ds = c.fromDs AND w.step < c.step) ELSE 0 END AS fromVer
349 , c.fromVar, c.toDs
350 , (SELECT COUNT(*) FROM work._lin_writes w WHERE w.ds = c.toDs AND w.step <= c.step) AS toVer
351 , c.toVar, c.expr
352 FROM work._lin_c1 c
353 LEFT JOIN &&p_stepsDsName s ON s.step = c.step
354 ORDER BY c.step, c.toDs, c.toVar, c.method, c.fromDs, c.fromVar;
355 QUIT;
