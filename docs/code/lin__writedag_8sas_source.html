scaLineage
Custom built SAS macro programs to trigger recording and extracting information about input, output, and the use of macro symbols from a SAS program while it is running. The captured information can provide a simple static data lineage diagram. As a minimum, Base SAS Software, version 9.4 Maintenance 4 installation is required to run these macros.

1/**
2 @file lin_writedag.sas
3 @brief Draws the lineage of one SCAPROC recording as Graphviz (*.dot) DAGs.
4 @details Writes up to three files into p_outDir (render them with Graphviz, e.g. dot -Tsvg x.dot -o x.svg):
5
6 <ul>
7 <li><p_name>_dataset.dot data set level DAG: files, data sets and steps (data sets are cylinders, no columns);
8 with p_attrs_yn=Y it links to <p_name>_dataset_attrs.svg</li>
9 <li><p_name>_dataset_attrs.dot (only when p_attrs_yn=Y) the same DAG with the columns listed inside the data set
10 nodes; links to <p_name>_column.svg when the column DAG is written</li>
11 <li><p_name>_column.dot column level DAG: one box per column, grouped in a cluster per data set,
12 one arrow per lineage edge (colour = how the column was derived)</li>
13 </ul>
14
15 Colours in the column DAG: grey = same name (NAME), blue = expression (EXPR), purple = rename,
16 green = read from a file, red outline = unresolved (the step has source, but it does not say where the column came
17 from), orange outline = the step has no recorded source (NOSOURCE).
18 A data set written more than once appears once per version, e.g. WORK.ONE (v2).
19
20 Needs the tables of %lin_readLog and %lin_buildLineage.
21 Example usage:
22
23 %lin_writeDag(p_prefix=lin
24 , p_outDir=
25 , p_name=
26 , p_attrs_yn=Y
27 , p_includeSteps_yn=Y

36 , p_cmemDsName=work.&p_prefix._cmem
37 , p_collinDsName=work.&p_prefix._collin)
38
39 @param [in] p_prefix= Prefix of the WORK tables. Default:lin.
40 @param [in] p_outDir= Output folder. Required.
41 @param [in] p_name= Base name of the output files. Required.
42 @param [in] p_attrs_yn= Y|N also write <p_name>_dataset_attrs.dot, the dataset DAG with the columns listed. Default:N.
43 @param [in] p_includeSteps_yn= Y|N draw the processing steps (with run time) in the dataset DAG. Default:Y.
44 @param [in] p_columnDag_yn= Y|N also write the column DAG. Default:Y.
45 @param [in] p_rankDir= TB|BT|LR|RL layout direction. Default:LR.
46 @param [in] p_stepsDsName=Step information for each step. Default: work.&p_prefix._steps
47 @param [in] p_dsioDsName=Dataset input/output information for each step. Default: work.&p_prefix._dsio
48 @param [in] p_fileioDsName=File input/output information for each step. Default: work.&p_prefix._fileio
49 @param [in] p_attrsDsName=Attribute information for each step. Default: work.&p_prefix._attrs
50 @param [in] p_srcDsName=Source code for each step. Default: work.&p_prefix._src
51 @param [in] p_cmemDsName=Column memory information for each step. Default: work.&p_prefix._cmem
52 @param [in] p_collinDsName=Column lineage information for each step. Default: work.&p_prefix._collin

122 seq = 1; line = '}'; OUTPUT;
123 RUN;

125 %if (&p_cols = Y) %then

280 /* ---- columns of every data set version (from the ATTR records) ---- */
281 PROC SQL;
282 CREATE TABLE work._lin_cols AS
283 SELECT v.ds, v.ver, a.var, a.type, a.ord
284 FROM &p_attrsDsName a
285 INNER JOIN &p_dsverDsName v
286 ON v.step = a.step AND v.ds = a.ds AND ((a.role = 'INPUT' AND v.io = 'IN') OR (a.role = 'OUTPUT' AND v.io = 'OUT'))
287 ORDER BY v.ds, v.ver, a.var, a.ord;
288 QUIT;
289 DATA work._lin_cols;
290 SET work._lin_cols;
291 BY ds ver var;
292 IF (FIRST.var);
293 RUN;
294 PROC SORT DATA=work._lin_cols;
295 BY ds ver ord;
296 RUN;

298 /* ---- data set nodes: one per (ds, version); the version is only shown when a data set has several ---- */
299 PROC SQL;
300 CREATE TABLE work._lin_dsn AS
301 SELECT DISTINCT d.ds, d.ver, n.nver
302 FROM &p_dsverDsName d
303 INNER JOIN (SELECT ds, COUNT(DISTINCT ver) AS nver FROM &p_dsverDsName GROUP BY ds) n ON n.ds = d.ds
304 ORDER BY d.ds, d.ver;
305 QUIT;
306 DATA work._lin_dsn;
307 SET work._lin_dsn;
308 LENGTH id label $200;
309 id = CAT(STRIP(ds), '#', STRIP(PUT(ver, best12.))); /* node id: no blanks or brackets */
310 IF (nver = 1) THEN label = ds;
311 ELSE label = CAT(STRIP(ds), ' (v', STRIP(PUT(ver, best12.)), ')');
312 IF (nver > 1 AND ver = 0) THEN label = CAT(STRIP(ds), ' (existing)');
313 RUN;

316 /* the plain DAG never lists the columns; with p_attrs_yn=Y it links to a second DAG that does,

331 PROC SQL;
332 /* every column that is mentioned: attribute records, plus both ends of every lineage edge */
333 CREATE TABLE work._lin_cn AS
334 SELECT ds, ver, var FROM work._lin_cols
335 UNION
336 SELECT toDs, toVer, toVar FROM &p_collinDsName
337 UNION
338 SELECT fromDs, fromVer, fromVar FROM &p_collinDsName WHERE fromKind = 'DS';
339
340 CREATE TABLE work._lin_cn2 AS
341 SELECT c.ds, c.ver, c.var, COALESCE(n.label, CAT(STRIP(c.ds), ' (v', STRIP(PUT(c.ver, best12.)), ')')) AS label LENGTH=200
342 , CASE WHEN EXISTS (SELECT 1 FROM &p_collinDsName x
343 WHERE x.method = 'UNRESOLVED' AND x.toDs = c.ds AND x.toVer = c.ver AND x.toVar = c.var)
344 THEN 'red'
345 WHEN EXISTS (SELECT 1 FROM &p_collinDsName x
346 WHERE x.method = 'NOSOURCE' AND x.toDs = c.ds AND x.toVer = c.ver AND x.toVar = c.var)
347 THEN 'darkorange' ELSE 'gray40' END AS colr LENGTH=10
348 , o.ord
349 FROM work._lin_cn c
350 LEFT JOIN work._lin_dsn n ON n.ds = c.ds AND n.ver = c.ver
351 LEFT JOIN work._lin_cols o ON o.ds = c.ds AND o.ver = c.ver AND o.var = c.var
352 ORDER BY c.ds, c.ver, o.ord, c.var;

354 /* file / in-stream sources */
355 CREATE TABLE work._lin_src AS
356 SELECT DISTINCT fromKind, fromDs, step FROM &p_collinDsName WHERE fromKind IN ('FILE', 'INLINE');
357 QUIT;
358
359 DATA work._lin_c2;
360 SET work._lin_cn2;
361 BY ds ver;
362 LENGTH line $4000 id $400;
363 RETAIN seq 0 cl 0;
364 sec = 2;
365 IF (FIRST.ver) THEN
366 DO;
367 cl + 1;
368 seq + 1;
369 line = CAT(' subgraph "cluster_', STRIP(PUT(cl, best12.)), '" {');
370 OUTPUT;
371 seq + 1;
372 line = CAT(' label="', %inner_lin_q(STRIP(label)), '"; style="filled,rounded"; fillcolor=lightblue; color=steelblue; fontsize=14;');
373 OUTPUT;
374 END;
375 id = CATS(ds, '#', ver, '.', var);
376 seq + 1;
377 line = CAT(' "', %inner_lin_q(STRIP(id)), '" [shape=box style=filled fillcolor=white color=', STRIP(colr), ' label="', %inner_lin_q(STRIP(var)), '"];');
378 OUTPUT;
379 IF (LAST.ver) THEN
380 DO;
381 seq + 1;
382 line = ' }';
383 OUTPUT;
384 END;
385 KEEP sec seq line;
386 RUN;

388 DATA work._lin_c3;
389 SET work._lin_src;
390 LENGTH line $4000;
391 sec = 3;
392 seq = _n_;
393 IF (fromKind = 'FILE') THEN
394 line = CAT(' "file:', %inner_lin_q(STRIP(fromDs)), '" [shape="note" fillcolor=lightgreen style=filled label="', %inner_lin_q(STRIP(SCAN(fromDs, -1, '/\'))), '" tooltip="', %inner_lin_q(STRIP(fromDs)), '"];');
395 ELSE
396 line = CAT(' "inline:', STRIP(PUT(step, best12.)), '" [shape="note" fillcolor=lightgreen style=filled label="in-stream data (step ', STRIP(PUT(step, best12.)), ')"];');
397 KEEP sec seq line;
398 RUN;
399
400 DATA work._lin_c4;
401 SET &p_collinDsName (WHERE=(method NE 'UNRESOLVED' AND fromKind NE 'NONE'));
402 LENGTH line $4000 src dst $400 colr $20 tip $1000;
403 sec = 5;
404 seq = _n_;
405 IF (fromKind = 'DS') THEN src = CATS(fromDs, '#', fromVer, '.', fromVar);
406 ELSE IF (fromKind = 'FILE') THEN src = CATS('file:', fromDs);
407 ELSE src = CATS('inline:', step);
408 dst = CATS(toDs, '#', toVer, '.', toVar);
409 SELECT (method);
410 WHEN ('EXPR') colr = 'blue';
411 WHEN ('RENAME') colr = 'purple';
412 WHEN ('NAME') colr = 'gray50';
413 OTHERWISE colr = 'darkgreen';
414 END;

418 KEEP sec seq line;

423 sec = 1;

427 seq = 4; line = ' node [fontname="Helvetica" fontsize=11];'; OUTPUT;
