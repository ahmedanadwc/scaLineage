scaLineage
Custom built SAS macro programs to trigger recording and extracting information about input, output, and the use of macro symbols from a SAS program while it is running. The captured information can provide a simple static data lineage diagram. As a minimum, Base SAS Software, version 9.4 Maintenance 4 installation is required to run these macros.

1/**
2 @file lin_exprsql.sas
3 @brief Extracts column derivations from the recorded source of PROC SQL steps.
4 @details Understands CREATE TABLE|VIEW x AS SELECT ... and INSERT INTO x SELECT ..., including joins,
5 table aliases, CASE expressions, calculated columns and UNION / INTERSECT / EXCEPT.
6 Creates
7
8 <ul><li>work.&p_prefix._exprSql: Column derivations from the recorded source of PROC SQL steps</li>
9|Variable|Description|
10|---|---|
11|step|The step number of the SQL step|
12|qid|The query ID (one per SELECT segment)|
13|toRaw|The raw text of the target (output) column|
14|toVar|The name of the target (output) column|
15|pos|The position of the column in the output|
16|kind|The kind of derivation (always EXPR)|
17|expr|The expression that defines the column|
18
19<li>work.&p_prefix._alias: Concatenated Library alias information</li>
20|Variable|Description|
21|---|---|
22|step|The step number of the SQL step|
23|qid|The query ID (one per SELECT segment)|
24|alias|The alias of the table|
25|tblRaw|The raw text of the table|
26</ul>
27
28 Example usage:
29
30 %lin_exprSql(p_prefix=&p_prefix, p_stmtsDsName=work.&p_prefix._stmts
31 , p_stepsDsName=work.&p_prefix._steps)
32
33 qid numbers the query segments of a step (a UNION has one qid per SELECT). Select items of the
34 second and later segments take the column name of the item at the same position in the first segment.
35 SELECT * / a.* items produce no row: they are covered by matching the column names.
36
37 @note Not handled: nested sub-queries in the FROM clause are treated as unknown tables (the columns
38 are then matched by name against every input table).
39
40 @param [in] p_prefix= Prefix of the WORK tables. Default:lin.
41 @param [in] p_stmtsDsName=Source code for each step. Default: work.&p_prefix._stmts
42 @param [in] p_stepsDsName=Step information for each step. Default: work.&p_prefix._steps
43
44
45 <h4> Data Inputs </h4>
46 @li work.&p_prefix._stmts
47 @li work.&p_prefix._steps
48
49
50 <h4> Data Outputs </h4>
51 @li work.&p_prefix._exprSql
52 @li work.&p_prefix._alias
53
54
55 @version 9.4
56 @author Ahmed Al-Attar
57**/

59%macro lin_exprSql(p_prefix=lin
60, p_stmtsDsName=work.&p_prefix._stmts
61, p_stepsDsName=work.&p_prefix._steps);
62
63 PROC SQL;
64 CREATE TABLE work._lin_sqlstmts AS
65 SELECT a.step, a.stmtno, a.text
66 FROM &p_stmtsDsName a
67 INNER JOIN &p_stepsDsName b ON a.step = b.step
68 WHERE b.procname = 'SQL'
69 ORDER BY a.step, a.stmtno;
70 QUIT;
71
72 DATA &p_prefix._exprSql (KEEP=step qid toRaw toVar pos kind expr)
73 &p_prefix._alias (KEEP=step qid alias tblRaw);
74 SET work._lin_sqlstmts;
75 BY step;
76 LENGTH toRaw $200 toVar $64 kind $8 expr $4000 alias $64 tblRaw $200
77 query seg msk msk2 mk txt selList fromTxt mfrom piece mpiece item mitem rest mrest $32000
78 q $1 ch $1 tbl $200 w1 w2 w3 $64 a1_1-a1_200 $64;
79 RETAIN qid 0 a1_1-a1_200 '' rxCreate rxInsert rxSet rxSelect rxFrom rxFromEnd rxJoin rxOn rxMod rxAs rxSimple rxInto;
80 ARRAY a1 {200} a1_1-a1_200;
81
82 IF (_n_ = 1) THEN
83 DO;
84 rxCreate = PRXPARSE('/^create\s+(?:table|view)\s+(\S+?)\s+as\s+(.*)$/i');
85 rxInsert = PRXPARSE('/^insert\s+into\s+(\S+?)(?:\s*\([^)]*\))?\s+(select\b.*)$/i');
86 rxSet = PRXPARSE('/\b(?:union|intersect|except)\b(?:\s+(?:all|distinct|corr|corresponding))*(?:\s+outer)?/i');
87 rxSelect = PRXPARSE('/^\s*select\s+(?:distinct\s+|all\s+|unique\s+)?/i');
88 rxFrom = PRXPARSE('/\sfrom\s/i');
89 rxFromEnd = PRXPARSE('/\s(?:where|group\s+by|having|order\s+by)\s/i');
90 rxJoin = PRXPARSE('/,|\b(?:natural\s+)?(?:(?:inner|left|right|full|cross)\s+)?(?:outer\s+)?join\b/i');
91 rxOn = PRXPARSE('/\bon\b/i');
92 rxMod = PRXPARSE('s/\s+(?:label|format|informat|length)\s*=\s*(?:''[^'']*''|"[^"]*"|\S+)//i');
93 rxAs = PRXPARSE('/\s+as\s+(\w+|''[^'']*''n|"[^"]*"n)\s*$/i');
94 rxSimple = PRXPARSE('/^(?:calculated\s+)?(?:\w+\.)?(\w+)$/i');
95 rxInto = PRXPARSE('s/\s+into\s+:.*?(?=\sfrom\s)//i');
96 END;
97 IF (FIRST.step) THEN
98 DO;
99 qid = 0;
100 DO i = 1 TO 200; a1{i} = ''; END;
101 END;
102
103 /* which statement is it? */
104 query = '';
105 toRaw = '';
106 IF (PRXMATCH(rxCreate, text)) THEN
107 DO;
108 toRaw = PRXPOSN(rxCreate, 1, text);
109 query = PRXPOSN(rxCreate, 2, text);
110 END;
111 ELSE IF (PRXMATCH(rxInsert, text)) THEN
112 DO;
113 toRaw = PRXPOSN(rxInsert, 1, text);
114 query = PRXPOSN(rxInsert, 2, text);
115 END;
116 IF (MISSING(query) OR NOT PRXMATCH('/^\s*\‍(?\s*select\b/i', query)) THEN RETURN;
117 query = PRXCHANGE(rxInto, 1, query);
118
119 /* mask: quoted text and everything inside parentheses becomes '_' (same length) */
120 txt = query; LINK mask; msk = mk;
121
122 /* split on top level set operators; segments are processed in turn */
123 segStart = 1;
124 segNo = 0;
125 start = 1; stop = LENGTH(msk);
126 IF (start > stop) THEN sp = 0;
127 ELSE CALL PRXNEXT(rxSet, start, stop, msk, sp, sl);
128 DO WHILE (1);
129 IF (sp > 0) THEN seg = SUBSTR(query, segStart, sp - segStart);
130 ELSE seg = SUBSTR(query, segStart);
131 segNo + 1;
132 qid + 1;
133 LINK doSegment;
134 IF (sp <= 0) THEN LEAVE;
135 segStart = sp + sl;
136 IF (start > stop) THEN sp = 0;
137 ELSE CALL PRXNEXT(rxSet, start, stop, msk, sp, sl);
138 END;
139 RETURN;
140
141 /* ---------- one SELECT ... FROM ... segment ---------- */
142 doSegment:
143 seg = STRIP(PRXCHANGE('s/\bouter\s*$//i', 1, STRIP(seg)));
144 /* tolerate a segment wrapped in one pair of brackets */
145 IF (seg =: '(') THEN
146 DO;
147 seg = STRIP(SUBSTR(seg, 2));
148 IF (SUBSTR(seg, LENGTH(seg)) = ')') THEN seg = STRIP(SUBSTR(seg, 1, LENGTH(seg) - 1));
149 END;
150 txt = seg; LINK mask; msk2 = mk;
151
152 IF (NOT PRXMATCH(rxSelect, seg)) THEN RETURN;
153 CALL PRXSUBSTR(rxSelect, seg, sPos, sLen);
154 fp = PRXMATCH(rxFrom, msk2);
155 IF (fp = 0) THEN
156 DO; /* no FROM: constants only */
157 selList = SUBSTR(seg, sPos + sLen);
158 fromTxt = '';
159 END;
160 ELSE
161 DO;
162 selList = SUBSTR(seg, sPos + sLen, fp - (sPos + sLen) + 1);
163 rest = SUBSTR(seg, fp + 6);
164 mrest = SUBSTR(msk2, fp + 6);
165 ep = PRXMATCH(rxFromEnd, mrest);
166 IF (ep > 0) THEN fromTxt = SUBSTR(rest, 1, ep);
167 ELSE fromTxt = rest;
168 END;
169
170 /* --- tables of the FROM clause --- */
171 IF (NOT MISSING(fromTxt)) THEN
172 DO;
173 txt = fromTxt; LINK mask; mfrom = mk;
174 pStart = 1;
175 jStart = 1; jStop = LENGTH(mfrom);
176 IF (jStart > jStop) THEN jp = 0;
177 ELSE CALL PRXNEXT(rxJoin, jStart, jStop, mfrom, jp, jl);
178 DO WHILE (1);
179 IF (jp > 0) THEN piece = SUBSTR(fromTxt, pStart, jp - pStart);
180 ELSE piece = SUBSTR(fromTxt, pStart);
181 LINK doPiece;
182 IF (jp <= 0) THEN LEAVE;
183 pStart = jp + jl;
184 IF (jStart > jStop) THEN jp = 0;
185 ELSE CALL PRXNEXT(rxJoin, jStart, jStop, mfrom, jp, jl);
186 END;
187 END;
188
189 /* --- select items --- */
190 txt = selList; LINK mask; mitem = mk;
191 k = 0; iStart = 1; n = LENGTH(selList);
192 DO i = 1 TO n + 1;
193 IF (i = n + 1 OR SUBSTR(mitem, i, 1) = ',') THEN
194 DO;
195 item = STRIP(SUBSTR(selList, iStart, i - iStart));
196 iStart = i + 1;
197 IF (NOT MISSING(item)) THEN
198 DO;
199 k + 1;
200 LINK doItem;
201 END;
202 END;
203 END;
204 RETURN;
205
206 /* ---------- one table of the FROM clause (item in piece) ---------- */
207 doPiece:
208 piece = STRIP(piece);
209 IF (MISSING(piece)) THEN RETURN;
210 txt = piece; LINK mask; mpiece = mk;
211 onp = PRXMATCH(rxOn, mpiece);
212 IF (onp > 0) THEN piece = STRIP(SUBSTR(piece, 1, onp - 1));
213 IF (MISSING(piece)) THEN RETURN;
214 alias = '';
215 IF (piece =: '(') THEN
216 DO; /* sub-query: unknown table, alias follows the closing bracket */
217 tbl = '(subquery)';
218 d = 0;
219 DO j = 1 TO LENGTH(piece);
220 ch = SUBSTR(piece, j, 1);
221 IF (ch = '(') THEN d + 1;
222 ELSE IF (ch = ')') THEN
223 DO;
224 d + (-1);
225 IF (d = 0) THEN LEAVE;
226 END;
227 END;
228 rest = STRIP(SUBSTR(piece, MIN(j + 1, LENGTH(piece) + 1)));
229 END;
230 ELSE
231 DO;
232 tbl = SCAN(piece, 1, ' (');
233 j = INDEX(piece, STRIP(tbl)) + LENGTH(STRIP(tbl));
234 rest = STRIP(SUBSTR(piece, j));
235 IF (rest =: '(') THEN
236 DO; /* skip dataset options */
237 d = 0;
238 DO j = 1 TO LENGTH(rest);
239 ch = SUBSTR(rest, j, 1);
240 IF (ch = '(') THEN d + 1;
241 ELSE IF (ch = ')') THEN
242 DO;
243 d + (-1);
244 IF (d = 0) THEN LEAVE;
245 END;
246 END;
247 rest = STRIP(SUBSTR(rest, MIN(j + 1, LENGTH(rest) + 1)));
248 END;
249 END;
250 w1 = SCAN(rest, 1, ' ');
251 w2 = SCAN(rest, 2, ' ');
252 IF (LOWCASE(w1) = 'as') THEN alias = w2;
253 ELSE IF (NOT MISSING(w1)) THEN alias = w1;
254 IF (MISSING(alias) AND tbl NE '(subquery)') THEN alias = SCAN(tbl, -1, '.');
255 alias = UPCASE(alias);
256 tblRaw = tbl;
257 OUTPUT &p_prefix._alias;
258 /* a table can also be referred to by its own name */
259 IF (tbl NE '(subquery)' AND UPCASE(SCAN(tbl, -1, '.')) NE alias) THEN
260 DO;
261 alias = UPCASE(SCAN(tbl, -1, '.'));
262 OUTPUT &p_prefix._alias;
263 END;
264 RETURN;
265
266 /* ---------- one select item (text in item, position k) ---------- */
267 doItem:
268 item = PRXCHANGE(rxMod, -1, item);
269 item = STRIP(item);
270 alias = '';
271 IF (PRXMATCH(rxAs, item)) THEN
272 DO;
273 alias = PRXPOSN(rxAs, 1, item);
274 /* name literal: 'my col'n -> my col */
275 IF (PRXMATCH('/^["'']/', alias)) THEN alias = PRXCHANGE('s/^["'']([^"'']*)["'']n$/$1/i', 1, alias);
276 CALL PRXSUBSTR(rxAs, item, ap, al);
277 expr = STRIP(SUBSTR(item, 1, ap - 1));
278 END;
279 ELSE
280 DO;
281 expr = item;
282 IF (PRXMATCH(rxSimple, item)) THEN alias = PRXPOSN(rxSimple, 1, item);
283 END;
284 /* SELECT * and a.* are covered by name matching */
285 IF (expr = '*' OR PRXMATCH('/\.\s*\*$/', expr)) THEN RETURN;
286
287 IF (segNo = 1) THEN
288 DO;
289 IF (k <= 200) THEN a1{k} = alias;
290 END;
291 ELSE IF (k <= 200) THEN alias = a1{k};
292
293 toVar = alias;
294 pos = k;
295 kind = 'EXPR';
296 OUTPUT &p_prefix._exprSql;
297 RETURN;
298
299 /* ---------- mask: txt -> mk ---------- */
300 mask:
301 mk = txt;
302 q = '';
303 d = 0;
304 DO m = 1 TO LENGTH(txt);
305 ch = SUBSTR(txt, m, 1);
306 IF (q NE '') THEN
307 DO;
308 IF (ch = q) THEN q = '';
309 SUBSTR(mk, m, 1) = '_';
310 END;
311 ELSE IF (ch IN ("'", '"')) THEN
312 DO;
313 q = ch;
314 SUBSTR(mk, m, 1) = '_';
315 END;
316 ELSE IF (ch = '(') THEN
317 DO;
318 d + 1;
319 SUBSTR(mk, m, 1) = '_';
320 END;
321 ELSE IF (ch = ')') THEN
322 DO;
323 d + (-1);
324 SUBSTR(mk, m, 1) = '_';
325 END;
326 ELSE IF (d > 0) THEN SUBSTR(mk, m, 1) = '_';
327 END;
328 RETURN;
329 RUN;
330
331 PROC DATASETS LIB=WORK NOLIST NOWARN;
332 DELETE _lin_sqlstmts;
333 RUN; QUIT;
334
335%mend lin_exprSql;
