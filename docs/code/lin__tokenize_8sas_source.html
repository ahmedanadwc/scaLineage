scaLineage
Custom built SAS macro programs to trigger recording and extracting information about input, output, and the use of macro symbols from a SAS program while it is running. The captured information can provide a simple static data lineage diagram. As a minimum, Base SAS Software, version 9.4 Maintenance 4 installation is required to run these macros.

1/**
2 @file lin_tokenize.sas
3 @brief Turns the derivation expressions into column references.
4 @details Reads work.&p_prefix._exprDs and work.&p_prefix._exprSql and creates
5
6 <ul>
7 <li>work.&p_prefix._eexpr: Expression information for each step.</li>
8 |Variable|Description|
9 |---|---|
10 |eid|Expression ID|
11 |step|Step number|
12 |qid|Query ID|
13 |toRaw|Target raw column|
14 |toVar|Target variable|
15 |pos|Position in the expression|
16 |kind|Type of expression|
17 |expr|Expression text|
18 |qual|Qualifier (alias or data set)|
19 |ident|Column identifier|
20 |calc|Flag indicating if the column is calculated|
21
22 <li>work.&p_prefix._ref: Column references for each expression.</li>
23 |Variable|Description|
24 |---|---|
25 |eid|Expression ID|
26 |qual|Qualifier (alias or data set)|
27 |ident|Column identifier|
28 |calc|Flag indicating if the column is calculated|
29 </ul>
30
31 An expression is the right hand side of one assignment or select item. A reference is a column
32 name used inside it. Quoted strings, function calls
33 (name followed by a bracket), formats (name followed by a dot), macro variables, numbers and
34 language keywords are not references. 'alias.column' keeps the alias in QUAL, and a column that
35 follows the SQL keyword CALCULATED has CALC=1. RENAME rows give one reference to the old name; its QUAL is the data set of the rename= option.
36
37 Example usage:
38
39 %lin_tokenize(p_prefix=lin)
40
41 @param [in] p_prefix= Prefix of the WORK tables. Default:lin.
42 @param [in] p_exprSqlDsName=Expression information for each SQL step. Default: work.&p_prefix._exprSql
43 @param [in] p_exprDsDsName=Expression information for each DATA step. Default: work.&p_prefix._exprDs
44
45
46 <h4> Data Inputs </h4>
47 @li work.&p_prefix._exprSql
48 @li work.&p_prefix._exprDs
49
50
51 <h4> Data Outputs </h4>
52 @li work.&p_prefix._eexpr
53 @li work.&p_prefix._ref
54
55
56 @version 9.4
57 @author Ahmed Al-Attar
58**/

60%macro lin_tokenize(p_prefix=lin
61, p_exprSqlDsName=work.&p_prefix._exprSql
62, p_exprDsDsName=work.&p_prefix._exprDs);
63
64 DATA &p_prefix._eexpr (KEEP=eid step qid toRaw toVar pos kind expr)
65 &p_prefix._ref (KEEP=eid qual ident calc);
66
67 LENGTH fromRaw $200;
68 SET &p_exprDsDsName &p_exprSqlDsName;
69 LENGTH e $4000 qual ident $64 nc pc $1 kw $600;
70 RETAIN rxLit1 rxLit2 rxTok kw;
71 IF (_n_ = 1) THEN
72 DO;
73 rxLit1 = PRXPARSE("s/'[^']*'/ /");
74 rxLit2 = PRXPARSE('s/"[^"]*"/ /');
75 rxTok = PRXPARSE('/(?:([A-Za-z_]\w*)\s*\.\s*)?([A-Za-z_]\w*)/');
76 kw = 'AND OR NOT IN IS NULL MISSING LIKE BETWEEN DISTINCT FROM SELECT ALL ANY EXISTS SOME CASE WHEN THEN ELSE END AS '
77 || 'TRUE FALSE IF DO TO BY WHILE UNTIL EQ NE LT GT LE GE _N_ _ERROR_ _ALL_ _CHARACTER_ _NUMERIC_ _TEMPORARY_ '
78 || 'CONTAINS ASC DESC OF ESCAPE OUTER INNER LEFT RIGHT FULL JOIN ON WHERE GROUP ORDER HAVING UNION';
79 END;
80
81 eid = _n_;
82 OUTPUT &p_prefix._eexpr;
83 calc = 0;
84 qual = '';
85 IF (kind = 'RENAME') THEN
86 DO;
87 ident = expr;
88 qual = fromRaw; /* data set carrying the rename= option, if known */
89 OUTPUT &p_prefix._ref;
90 RETURN;
91 END;
92 IF (kind NE 'EXPR') THEN RETURN;
93
94 e = expr;
95 CALL PRXCHANGE(rxLit1, -1, e);
96 CALL PRXCHANGE(rxLit2, -1, e);
97 start = 1;
98 stop = LENGTH(e);
99 calcNext = 0;
100 IF (start > stop) THEN p = 0;
101 ELSE CALL PRXNEXT(rxTok, start, stop, e, p, l);
102 DO WHILE (p > 0);
103 qual = PRXPOSN(rxTok, 1, e);
104 ident = PRXPOSN(rxTok, 2, e);
105 j = p + l;
106 DO WHILE (j <= stop AND SUBSTR(e, j, 1) = ' '); j + 1; END;
107 nc = '';
108 IF (j <= stop) THEN nc = SUBSTR(e, j, 1);
109 pc = '';
110 IF (p > 1) THEN pc = SUBSTR(e, p - 1, 1);
111 use = 1;
112 IF (pc IN ('&', ':') OR (pc >= '0' AND pc <= '9')) THEN use = 0;
113 ELSE IF (nc = '(') THEN use = 0; /* function call / array element */
114 ELSE IF (nc = '.' AND MISSING(qual)) THEN use = 0; /* format such as z4. or first. */
115 ELSE IF (MISSING(qual) AND UPCASE(ident) = 'CALCULATED') THEN
116 DO;
117 use = 0;
118 calcNext = 1;
119 END;
120 ELSE IF (MISSING(qual) AND FINDW(kw, STRIP(ident), ' ', 'it') > 0) THEN use = 0;
121 IF (use) THEN
122 DO;
123 IF (UPCASE(qual) IN ('FIRST', 'LAST')) THEN qual = '';
124 calc = calcNext;
125 OUTPUT &p_prefix._ref;
126 calcNext = 0;
127 END;
128 IF (start > stop) THEN p = 0;
129 ELSE CALL PRXNEXT(rxTok, start, stop, e, p, l);
130 END;
131 RUN;
132
133%mend lin_tokenize;
