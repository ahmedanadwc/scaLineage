scaLineage
Custom built SAS macro programs to trigger recording and extracting information about input, output, and the use of macro symbols from a SAS program while it is running. The captured information can provide a simple static data lineage diagram. As a minimum, Base SAS Software, version 9.4 Maintenance 4 installation is required to run these macros.

1/**
2 @file lin_writejson.sas
3 @brief Writes the lineage of one SCAPROC recording as a node / edge JSON graph.
4 @details One file holds both levels, so a consumer can filter on "level":
5
6 <ul>
7 <li>nodes</li>
8 {"id", "label", "type", "level", ...}<br/>
9 type "dataset" (level dataset; "columns": [{name, type}] when p_attrs_yn=Y),
10 "file", "step" (with "step" number), "column" (level column; "dataset" = id of its data set node, "unresolved": true when the step has source that does not say where the column came from, "nosource": true when the step has no recorded source)
11
12 <li>edges</li>
13 {"source", "target", "type", "level", ...}<br/>
14 level dataset: type "input" (to a step), "output" (from a step) or "flow" (no step nodes)<br/>
15 level column: type "column", "method" (EXPR|RENAME|NAME|FILE|INPUT|CONST), "step", "expr"<br/>
16 </ul>
17
18 @note Node ids are the ids used in the DOT files. A data set written more than once is one node per version.
19
20 Needs the tables of %lin_readLog and %lin_buildLineage.
21
22 @param [in] p_prefix= Prefix of the WORK tables. Default:lin.
23 @param [in] p_outFile= Full path of the JSON file. Required.
24 @param [in] p_name= Name stored in the "source" field. Required.
25 @param [in] p_attrs_yn= Y|N list the columns inside the data set nodes. Default:N.
26 @param [in] p_includeSteps_yn= Y|N step nodes with input -> step -> output edges. Default:Y.
27 @param [in] p_columnGraph_yn= Y|N include the column nodes and column edges. Default:Y.
28 @param [in] p_stepsDsName=Step information for each step. Default: work.&p_prefix._steps
29 @param [in] p_fileioDsName=File input/output information for each step. Default: work.&p_prefix._fileio
30 @param [in] p_attrsDsName=Attribute information for each step. Default: work.&p_prefix._attrs
31 @param [in] p_collinDsName=Column lineage information for each step. Default: work.&p_prefix._collin
32 @param [in] p_dsverDsName=Dataset version information for each step. Default: work.&p_prefix._dsver
33
34
35 <h4> Data Inputs </h4>
36 @li work.&p_prefix._steps
37 @li work.&p_prefix._fileio
38 @li work.&p_prefix._attrs
39 @li work.&p_prefix._collin
40 @li work.&p_prefix._dsver
41
42 @version 9.4
43 @author Ahmed Al-Attar
44**/

46/* expression that escapes a character value for a JSON string (backslash, quote, control characters) */
47%macro inner_lin_jq(p_x);
48 PRXCHANGE('s/[\x00-\x1F]+/ /', -1, TRANWRD(TRANWRD(&p_x, '\', '\\'), '"', '\"'))
49%mend inner_lin_jq;

75 /* ---- columns of every data set version (from the ATTR records) ---- */
76 PROC SQL;
77 CREATE TABLE work._lj_cols AS
78 SELECT v.ds, v.ver, a.var, a.type, a.ord
79 FROM &p_attrsDsName a
80 INNER JOIN &p_dsverDsName v
81 ON v.step = a.step AND v.ds = a.ds AND ((a.role = 'INPUT' AND v.io = 'IN') OR (a.role = 'OUTPUT' AND v.io = 'OUT'))
82 ORDER BY v.ds, v.ver, a.var, a.ord;
83 QUIT;
84 DATA work._lj_cols;
85 SET work._lj_cols;
86 BY ds ver var;
87 IF (FIRST.var);
88 RUN;
89 PROC SORT DATA=work._lj_cols;
90 BY ds ver ord;
91 RUN;

93 /* ---- data set nodes: one per (ds, version); the version is only shown when a data set has several ---- */
94 PROC SQL;
95 CREATE TABLE work._lj_dsn AS
96 SELECT DISTINCT d.ds, d.ver, n.nver
97 FROM &p_dsverDsName d
98 INNER JOIN (SELECT ds, COUNT(DISTINCT ver) AS nver FROM &p_dsverDsName GROUP BY ds) n ON n.ds = d.ds
99 ORDER BY d.ds, d.ver;
100 QUIT;
101 DATA work._lj_dsn;
102 SET work._lj_dsn;
103 LENGTH id label $200;
104 id = CAT(STRIP(ds), '#', STRIP(PUT(ver, best12.)));
105 IF (nver = 1) THEN label = ds;
106 ELSE label = CAT(STRIP(ds), ' (v', STRIP(PUT(ver, best12.)), ')');
107 IF (nver > 1 AND ver = 0) THEN label = CAT(STRIP(ds), ' (existing)');
108 RUN;

110 /* ===== nodes: sec 1 data sets (with columns), 2 files, 3 steps, 4 columns, 5 in-stream data ===== */
111 DATA work._lj_n1 (KEEP=sec obj seq st line);
112 MERGE work._lj_dsn (IN=isDs) work._lj_cols (KEEP=ds ver var type ord);
113 BY ds ver;
114 IF (NOT isDs) THEN DELETE;
115 LENGTH line $4000;
116 RETAIN obj 0 seq 0 nc 0 hasc 0;
117 sec = 1;
118 st = 0;
119 IF (FIRST.ver) THEN
120 DO;
121 obj + 1;
122 nc = 0;
123 hasc = ("&l_attrs" = 'Y' AND NOT MISSING(var));
124 line = CAT('{"id": "', %inner_lin_jq(STRIP(id)), '", "label": "', %inner_lin_jq(STRIP(label)), '", "type": "dataset", "level": "dataset"');
125 IF (hasc) THEN line = CAT(STRIP(line), ', "columns": [');
126 ELSE line = CAT(STRIP(line), '}');
127 st = 1;
128 seq + 1;
129 OUTPUT;
130 st = 0;
131 END;
132 IF (hasc AND NOT MISSING(var)) THEN
133 DO;
134 line = CAT(IFC(nc > 0, ',', ' '), '{"name": "', %inner_lin_jq(STRIP(var)), '", "type": "', IFC(type = 'CHARACTER', 'char', 'num'), '"}');
135 nc + 1;
136 seq + 1;
137 OUTPUT;
138 END;
139 IF (LAST.ver AND hasc) THEN
140 DO;
141 line = ']}';
142 seq + 1;
143 OUTPUT;
144 END;
145 RUN;

210 CREATE TABLE work._lj_cn2 AS
211 SELECT c.ds, c.ver, c.var
212 , EXISTS (SELECT 1 FROM &p_collinDsName x

264 CREATE TABLE work._lj_e1 AS
265 SELECT 1 AS sec, MONOTONIC() AS obj, 1 AS seq, 1 AS st
266 , CAT('{"source": "', %inner_lin_jq(STRIP(i.node)), '", "target": "', %inner_lin_jq(STRIP(o.node))
