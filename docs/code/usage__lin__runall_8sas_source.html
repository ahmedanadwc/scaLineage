scaLineage
Custom built SAS macro programs to trigger recording and extracting information about input, output, and the use of macro symbols from a SAS program while it is running. The captured information can provide a simple static data lineage diagram. As a minimum, Base SAS Software, version 9.4 Maintenance 4 installation is required to run these macros.

1/**
2 @file usage_lin_runAll.sas
3 @brief Example: extract data / column lineage and draw the DAGs for every SCAPROC *.txt in a folder.
4 @details Edit the three paths below, then submit the whole program in SAS.
5 Render the resulting files with Graphviz, e.g. dot -Tsvg x_column.dot -o x_column.svg
6**/
7
8/* ---- where things are --------------------------------------------------------------------- */
9%let g_linRoot = %sysfunc(sysget(HOME))/scaLineage; /* folder that holds code/ and your logs */
10%let g_logDir = &g_linRoot/logs; /* the SCAPROC *.txt files */
11%let g_outDir = &g_linRoot/output; /* results, one sub-folder per log */
12
13Options MPRINT FULLSTIMER MSGLEVEL=I;
14
15/* ---- load the macros ------------------------------------------------------------------------ */
16%macro loadLineage(p_codeDir=);
17 %LOCAL i f;
18 %let f = lin_startrecord lin_stoprecord lin_readlog lin_prepsource lin_exprdatastep lin_exprsql lin_tokenize lin_buildlineage
19 lin_writedag lin_writejson lin_processlog lin_runall;
20 %do i = 1 %to %sysfunc(countw(&f));
21 %include "&p_codeDir/%scan(&f, &i).sas";
22 %end;
23%mend loadLineage;
24%loadLineage(p_codeDir=&g_linRoot/code/macros)
25
26/* ---- make sure the output folder exists ------------------------------------------------------ */
27OPTIONS DLCREATEDIR;
28LIBNAME _out "&g_outDir";
29LIBNAME _out CLEAR;
30
31/* ---- every *.txt in the log folder ----------------------------------------------------------- */
32%lin_runAll(
33 p_logDir=&g_logDir
34, p_outDir=&g_outDir
35, p_attrs_yn=Y /* list the columns inside the data set nodes */
36, p_includeSteps_yn=Y /* draw the steps with their run times */
37, p_columnDag_yn=Y /* also draw the column level DAG */
38, p_rankDir=LR)
39
40/* work.lin_summary holds one row per log: steps, data sets, edges by kind, unresolved columns */
41PROC PRINT DATA=work.lin_summary NOOBS;
42RUN;
