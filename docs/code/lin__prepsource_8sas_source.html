Custom built SAS macro programs to trigger recording and extracting information about input, output, and the use of macro symbols from a SAS program while it is running. The captured information can provide a simple static data lineage diagram. As a minimum, Base SAS Software, version 9.4 Maintenance 4 installation is required to run these macros.

1/**
2 @file lin_prepsource.sas
3 @brief Splits the recorded source of every step into statements.
4 @details Reads &p_prefix._src (see %lin_readLog) and creates
5
6 <ul><li>work.&p_prefix._stmts: Source code for each step split into individual statements</li>
7 |Variable|Description|
8 |---|---|
9 |step|The step number of the statement|
10 |stmtno|The statement number within the step|
11 |text|The text of the statement (no trailing semicolon)|
12 </ul>
13
14 Example usage:
15
16 %lin_prepSource(p_prefix=&p_prefix, p_srcDsName=work.&p_prefix._src)
17
18 Cleaning applied before splitting: block comments are removed, in-stream data
19 (DATALINES/CARDS ... ;) is dropped, white space is collapsed, and semicolons inside
20 quoted strings do not end a statement.
21
22 Steps generated by a macro have no recorded source; they simply get no rows here, and the
23 lineage engines fall back to matching columns by name.
24
25 @param [in] p_prefix= Prefix of the WORK tables. Default:lin.
26 @param [in] p_srcDsName= Source code for each step. Default: work.&p_prefix._src
27
28 <h4> Data Inputs </h4>
29 @li work.&p_prefix._src
30
31
32 <h4> Data Outputs </h4>
33 @li work.&p_prefix._stmts
34

38**/

40%macro lin_prepSource(p_prefix=lin, p_srcDsName=work.&p_prefix._src);
41
42 /* 1. glue the source lines of each step into one string, dropping in-stream data lines */
43 DATA work._lin_oneline (KEEP=step src);
44 SET &p_srcDsName;
45 BY step;
46 LENGTH src $32000;
47 RETAIN src '' inData 0;
48 IF (FIRST.step) THEN
49 DO;
50 src = '';
51 inData = 0;
52 END;
53
54 IF (inData) THEN
55 DO;
56 /* in-stream data ends at a line that starts with a semicolon */
57 IF (STRIP(text) =: ';') THEN inData = 0;
58 END;
59 ELSE
60 DO;
61 src = CATX(' ', src, STRIP(TRANSLATE(text, ' ', '090D'x)));
62 IF (PRXMATCH('/\b(datalines|datalines4|cards|cards4)\s*;\s*$/i', text)) THEN inData = 1;
63 END;
64 IF (LAST.step) THEN OUTPUT;
65 RUN;
66
67 /* 2. strip block comments, then split into statements outside of quotes */
68 DATA &p_prefix._stmts (KEEP=step stmtno text);
69 SET work._lin_oneline;
70 LENGTH text $32000 q $1 ch $1;
71 RETAIN rxCmt;
72 IF (_n_ = 1) THEN rxCmt = PRXPARSE('s/\/\*.*?\*\// /');
73 CALL PRXCHANGE(rxCmt, -1, src);
74 src = COMPBL(src);
75
76 stmtno = 0;
77 q = '';
78 text = '';
79 k = 0; /* characters currently in text */
80 n = LENGTH(src);
81 DO i = 1 TO n;
82 ch = SUBSTR(src, i, 1);
83 IF (ch = ';' AND q = '') THEN
84 DO;
85 IF (k > 0 AND NOT MISSING(text)) THEN
86 DO;
87 stmtno + 1;
88 text = STRIP(text);
89 OUTPUT;
90 END;
91 text = '';
92 k = 0;
93 END;
94 ELSE
95 DO;
96 IF (q NE '') THEN
97 DO;
98 IF (ch = q) THEN q = '';
99 END;
100 ELSE IF (ch IN ("'", '"')) THEN q = ch;
101 k + 1;
102 SUBSTR(text, k, 1) = ch;
103 END;
104 END;
105 IF (k > 0 AND NOT MISSING(text)) THEN
106 DO;
107 stmtno + 1;
108 text = STRIP(text);
109 OUTPUT;
110 END;
111 RUN;
112
113 PROC DATASETS LIB=WORK NOLIST NOWARN;
114 DELETE _lin_oneline;
115 RUN; QUIT;
116
117%mend lin_prepSource;
